22def derive_generator(seed: int, *labels: object) -> torch.Generator: 

23 """CPU generator keyed on ``seed`` and any labels (metric name, arm, repeat...). 

24 

25 SHA-256 rather than ``hash`` keeps the derived seed stable across processes; 

26 the mask keeps it in torch's accepted int64 range. 

27 """ 

28 key = ":".join([str(seed), *(str(label) for label in labels)]).encode() 

29 derived = int.from_bytes(hashlib.sha256(key).digest()[:8], "little") & 0x7FFFFFFFFFFFFFFF 

30 return torch.Generator(device="cpu").manual_seed(derived)

60 """Percentile bootstrap interval of ``statistic`` along ``dim``, vectorized over the rest. 

61 

62 Resamples with replacement using a CPU ``generator`` (``derive_generator(0)`` 

63 when omitted), ``chunk_size`` resamples at a time so memory stays at 

64 ``chunk_size × n`` per leading cell. Returns ``(low, high)`` with ``dim`` 

65 removed. 

66 

67 A percentile bootstrap cannot express uncertainty a sample does not contain: 

68 a constant vector resamples to itself and the interval collapses to a point 

69 whatever ``n`` is, and intervals on 4-8 samples are wide and skewed. For 

70 binary success rates prefer an exact interval 

71 (:func:`~transformer_lens.tools.analysis.jacobian_lens_causal_swap_benchmark.success_rate_ci`). 

72 """

79 moved = _to_last(x, dim) 

80 n = moved.shape[-1]

82 raise ValueError("cannot bootstrap an empty axis")

84 reduce = torch.mean if statistic == "mean" else lambda t, dim: t.median(dim=dim).values 

85 stats = [] 

86 for start in range(0, n_resamples, chunk_size):

88 idx = torch.randint(0, n, (count, n), generator=gen).to(moved.device)

98def sign_flip_permutation_pvalue( 

99 effects: torch.Tensor, 

100 *, 

101 dim: int = -1, 

102 n_permutations: int = 1000, 

103 generator: Optional[torch.Generator] = None,

106) -> torch.Tensor: 

107 """Paired sign-flip permutation p-value that the mean effect along ``dim`` is zero. 

108 

109 The null is that each per-example effect is symmetric about zero, so every 

110 pattern of sign flips is equally likely. With ``n <= exact_max_n`` examples 

111 all ``2**n`` patterns are enumerated and the p-value is exact and 

112 deterministic; otherwise ``n_permutations`` random patterns are drawn from 

113 the CPU ``generator`` and the p-value is ``(hits + 1) / (n_permutations + 1)``. 

114 ``alternative`` is ``"two-sided"`` (``|null| >= |observed|``), ``"greater"`` 

115 or ``"less"``. Returns a tensor with ``dim`` removed.

120 raise ValueError("n_permutations must be >= 1")

124 raise ValueError("cannot permute an empty axis") 

125 observed = moved.mean(dim=-1, keepdim=True) # [..., 1]

127 signs = torch.tensor( 

128 list(itertools.product((1.0, -1.0), repeat=n)), dtype=moved.dtype, device=moved.device

132 return hits / signs.shape[0] 

133 gen = generator if generator is not None else derive_generator(0)
